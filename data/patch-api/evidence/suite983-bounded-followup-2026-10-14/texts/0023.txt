#[path = "old_spells.rs"]
mod old_catalog;
#[path = "<PRIVATE_PATH_ROOT_01>/data/spells.rs"]
mod new_catalog;

fn main() {
    let old = &old_catalog::SPELL_DB;
    let new = &new_catalog::SPELL_DB;
    assert!(!old.contains_key(&372610), "original must have the catalog gap");
    assert_eq!(new.len(), old.len() + 1, "exactly one added record");
    compare_existing_records();
    compare_sourced_record();
    println!("PASS: {} existing records unchanged; only 372610 added", old.len());
}

fn compare_existing_records() {
    for (id, before) in old_catalog::SPELL_DB.entries() {
        let after = new_catalog::get_spell(*id).expect("existing spell removed");
        let before_fields = (
            before.name, before.subtext, before.icon_file_data_id,
            before.school_mask, before.implicit_target,
        );
        let after_fields = (
            after.name, after.subtext, after.icon_file_data_id,
            after.school_mask, after.implicit_target,
        );
        assert_eq!(before_fields, after_fields, "existing spell {id} changed");
    }
}

fn compare_sourced_record() {
    let spell = new_catalog::get_spell(372610).expect("sourced spell missing");
    assert_eq!(spell.name, "Skyward Ascent");
    assert_eq!(spell.subtext, "Skyriding");
    assert_eq!(spell.icon_file_data_id, 4640498);
    assert_eq!(spell.school_mask, 1);
    assert_eq!(spell.implicit_target, 18);
}
